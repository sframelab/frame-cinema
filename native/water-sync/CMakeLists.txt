cmake_minimum_required(VERSION 3.17)
project(frame_water_sync LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE FILEPATH "" FORCE)
if(NOT GODOT_CPP_DIR OR NOT GODOTCPP_CUSTOM_API_FILE)
    message(FATAL_ERROR "Pinned godot-cpp and the actual 4.7.2 API are required")
endif()
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(frame_water_sync SHARED frame_water_sync.cpp)
target_link_libraries(frame_water_sync PRIVATE godot-cpp)
set_target_properties(frame_water_sync PROPERTIES PREFIX "lib" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
