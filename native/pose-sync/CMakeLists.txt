cmake_minimum_required(VERSION 3.17)
project(frame_pose_sync LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE FILEPATH "" FORCE)
if(NOT GODOT_CPP_DIR OR NOT GODOTCPP_CUSTOM_API_FILE)
    message(FATAL_ERROR "Specify the pinned bindings directory and the actual Godot 4.7.2 API dump")
endif()
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(frame_pose_sync SHARED frame_pose_sync.cpp)
target_link_libraries(frame_pose_sync PRIVATE godot-cpp)
target_compile_definitions(frame_pose_sync PRIVATE XR_USE_TIMESPEC)
set_target_properties(frame_pose_sync PROPERTIES PREFIX "lib" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
