cmake_minimum_required(VERSION 3.16)
project(frame_cinema_stereo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ncnn CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(check_strength_sequence ../check_strength_sequence.cpp)
target_include_directories(check_strength_sequence PRIVATE ..)
target_compile_options(check_strength_sequence PRIVATE -UNDEBUG)
target_link_libraries(check_strength_sequence PRIVATE ncnn OpenMP::OpenMP_CXX)
add_executable(check_depth_sequence ../check_depth_sequence.cpp)
target_compile_options(check_depth_sequence PRIVATE -UNDEBUG)
target_link_libraries(check_depth_sequence PRIVATE ncnn OpenMP::OpenMP_CXX)
add_executable(bench_depth bench_depth.cpp)
target_link_libraries(bench_depth PRIVATE ncnn OpenMP::OpenMP_CXX)
add_executable(test_capture test_capture.cpp)
target_link_libraries(test_capture PRIVATE OpenGL::GL X11 Xcomposite)
add_executable(test_stereo_gl test_stereo_gl.cpp)
target_link_libraries(test_stereo_gl PRIVATE OpenGL::GL X11 Xcomposite)
add_executable(test_policy test_policy.cpp)
target_compile_options(test_policy PRIVATE -UNDEBUG)
add_executable(test_depth_scale test_depth_scale.cpp)
target_compile_options(test_depth_scale PRIVATE -UNDEBUG)
target_link_libraries(test_depth_scale PRIVATE ncnn OpenMP::OpenMP_CXX)
add_executable(test_visual test_visual.cpp)
target_link_libraries(test_visual PRIVATE ncnn OpenMP::OpenMP_CXX OpenGL::GL X11 Xcomposite)
foreach(SIZE IN ITEMS 256 384 512)
  if(SIZE EQUAL 256)
    set(HEIGHT 160)
  elseif(SIZE EQUAL 384)
    set(HEIGHT 224)
  else()
    set(HEIGHT 288)
  endif()
  add_executable(bench_pipeline_${SIZE} bench_pipeline.cpp)
  target_compile_definitions(bench_pipeline_${SIZE} PRIVATE FRAME_CINEMA_DEPTH_WIDTH=${SIZE} FRAME_CINEMA_DEPTH_HEIGHT=${HEIGHT})
  target_link_libraries(bench_pipeline_${SIZE} PRIVATE ncnn OpenMP::OpenMP_CXX OpenGL::GL X11 Xcomposite)
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../panel_host.cpp")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AT REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0 cairo)
  add_executable(panel_host ../panel_host.cpp)
  target_include_directories(panel_host PRIVATE ..)
  target_link_directories(panel_host PRIVATE /opt/steamvr/bin/linuxarm64)
  target_link_libraries(panel_host PRIVATE ncnn OpenMP::OpenMP_CXX OpenGL::GL X11 Xcomposite Xtst openvr_api PkgConfig::AT pthread)
  # glslang and SPIRV-Tools ship in lib/ beside panel_host, because SteamOS updates change their versions.
  # An RPATH (not RUNPATH) also covers the libraries those load from each other.
  set_target_properties(panel_host PROPERTIES BUILD_RPATH "$ORIGIN/lib;/opt/steamvr/bin/linuxarm64")
  target_link_options(panel_host PRIVATE "LINKER:--disable-new-dtags")
  add_executable(test_live test_live.cpp)
  target_include_directories(test_live PRIVATE ..)
  target_link_directories(test_live PRIVATE /opt/steamvr/bin/linuxarm64)
  target_link_libraries(test_live PRIVATE ncnn OpenMP::OpenMP_CXX OpenGL::GL X11 Xcomposite openvr_api pthread)
  set_target_properties(test_live PROPERTIES BUILD_RPATH /opt/steamvr/bin/linuxarm64)
endif()
